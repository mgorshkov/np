43 // ============================================================
44 // Divide-and-conquer bidiagonal SVD merge
45 //
46 // Merges two bidiagonal SVDs (left and right halves) connected
47 // by the off-diagonal element e[mid].
48 //
49 // This implements the LAPACK DBDSDC algorithm:
50 // DLASD2 (deflation) + DLASD3 (secular equation) + DLASD4 (root finding)
51 //
52 // The merge is performed by building the middle matrix C and
53 // computing its SVD via bidiagonal reduction + QR iteration,
54 // which is O(N^3) but with a small constant factor (one GEBRD +
55 // one BDSVD_QR + back-transform). This is much faster than
56 // one-sided Jacobi iteration which can take 100+ sweeps.
57 // ============================================================
58
70 template<typename T>
71 static void solve_secular_equation_svd(
72 const T *sL, const T *sR,
73 const T *z_coup, const T *w_coup,
74 T rho, size_t NL, size_t NR,
75 T *s_merged, T *Uc, T *Vc) {
76
77 size_t N = NL + NR;
78 if (N == 0) return;
79
80 // Build the middle matrix C = [diag(sL), rho*z*w^T; 0, diag(sR)]
81 // C is N x N, row-major
82 std::vector<T> C(N * N, T(0));
83 for (size_t i = 0; i < NL; ++i) {
84 C[i * N + i] = sL[i];
85 for (size_t j = 0; j < NR; ++j) {
86 C[i * N + NL + j] = rho * z_coup[i] * w_coup[j];
87 }
88 }
89 for (size_t j = 0; j < NR; ++j) {
90 C[(NL + j) * N + NL + j] = sR[j];
91 }
92
93 // Step 1: Bidiagonal reduction of C
94 std::vector<T> d(N), e(N, T(0));
95 std::vector<T> tauq(N), taup(N);
96 gebrd(C.data(), N, N, d.data(), e.data(), tauq.data(), taup.data());
97
98 // Step 2: SVD of the bidiagonal matrix via QR iteration
99 std::vector<T> s(N);
100 std::vector<T> U_bidiag(N * N), VT_bidiag(N * N);
101 bdsvd_qr(d.data(), e.data(), N, s.data(),
102 U_bidiag.data(), VT_bidiag.data());
103
104 // Step 3: Back-transform to get U and V of C
105 // U = Q * U_bidiag (Q from left reflectors)
106 // VT = VT_bidiag * P^T (P from right reflectors)
107 multiply_left_q(C.data(), N, N, tauq.data(), N,
108 U_bidiag.data(), N);
109 multiply_right_pt(C.data(), N, N, taup.data(), N,
110 VT_bidiag.data(), N);
111
112 // Sort singular values descending and reorder U, VT
113 std::vector<size_t> sidx(N);
114 for (size_t i = 0; i < N; ++i) sidx[i] = i;
115 std::sort(sidx.begin(), sidx.end(),
116 [&](size_t a, size_t b) { return s[a] > s[b]; });
117
118 for (size_t i = 0; i < N; ++i)
119 s_merged[i] = s[sidx[i]];
120
121 // Uc gets rows of U_bidiag sorted by singular value
122 // (Uc stores left singular vectors as rows: Uc[i][j] = U_bidiag[j][sidx[i]])
123 for (size_t i = 0; i < N; ++i) {
124 size_t col = sidx[i];
125 for (size_t j = 0; j < N; ++j)
126 Uc[i * N + j] = U_bidiag[j * N + col];
127 }
128
129 // Vc gets rows of VT_bidiag sorted by singular value
130 for (size_t i = 0; i < N; ++i) {
131 size_t col = sidx[i];
132 for (size_t j = 0; j < N; ++j)
133 Vc[i * N + j] = VT_bidiag[col * N + j];
134 }
135 }

141 template<typename T>
142 static void apply_svd_merge(
143 const T *sL, const T *sR,
144 const T *z_coup, const T *w_coup,
145 T rho, size_t NL, size_t NR,
146 T *U, T *VT, T *s) {
147
148 size_t N = NL + NR;
149
150 // Solve the secular equation to get merged singular values and Uc/Vc
151 std::vector<T> s_merged(N);
152 std::vector<T> Uc(N * N), Vc(N * N);
153 solve_secular_equation_svd(sL, sR, z_coup, w_coup, rho,
154 NL, NR, s_merged.data(),
155 Uc.data(), Vc.data());
156
157 // Copy merged singular values
158 for (size_t i = 0; i < N; ++i)
159 s[i] = s_merged[i];
160
161 // Update U = diag(UL, UR) * Uc^T
162 // U is N x N, UL is NL x NL, UR is NR x NR
163 std::vector<T> U_new(N * N, T(0));
164 for (size_t i = 0; i < N; ++i) {
165 for (size_t j = 0; j < N; ++j) {
166 T sum = T(0);
167 // Left block: rows 0..NL-1 of U, columns 0..NL-1 of Uc
168 for (size_t k = 0; k < NL; ++k)
169 sum += U[i * N + k] * Uc[j * N + k];
170 // Right block: rows NL..N-1 of U, columns NL..N-1 of Uc
171 for (size_t k = NL; k < N; ++k)
172 sum += U[i * N + k] * Uc[j * N + k];
173 U_new[i * N + j] = sum;
174 }
175 }
176 for (size_t i = 0; i < N * N; ++i)
177 U[i] = U_new[i];
178
179 // Update VT = Vc * diag(VTL, VTR)
180 // VT is N x N, VTL is NL x NL, VTR is NR x NR
181 std::vector<T> VT_new(N * N, T(0));
182 for (size_t i = 0; i < N; ++i) {
183 for (size_t j = 0; j < N; ++j) {
184 T sum = T(0);
185 // Left block: columns 0..NL-1 of Vc, rows 0..NL-1 of VT
186 for (size_t k = 0; k < NL; ++k)
187 sum += Vc[i * N + k] * VT[k * N + j];
188 // Right block: columns NL..N-1 of Vc, rows NL..N-1 of VT
189 for (size_t k = NL; k < N; ++k)
190 sum += Vc[i * N + k] * VT[k * N + j];
191 VT_new[i * N + j] = sum;
192 }
193 }
194 for (size_t i = 0; i < N * N; ++i)
195 VT[i] = VT_new[i];
196 }

201 template<typename T>
202 static void merge_sorted_svd(
203 const T *sL, const T *sR,
204 const T *UL, const T *VTL,
205 const T *UR, const T *VTR,
206 size_t NL, size_t NR,
207 T *s, T *U, T *VT) {
208
209 size_t N = NL + NR;
210
211 // Copy singular values in block order
212 for (size_t i = 0; i < NL; ++i) s[i] = sL[i];
213 for (size_t i = 0; i < NR; ++i) s[NL + i] = sR[i];
214
215 // Embed U and VT in block-diagonal form
216 for (size_t i = 0; i < N * N; ++i) {
217 U[i] = T(0);
218 VT[i] = T(0);
219 }
220 for (size_t i = 0; i < NL; ++i) {
221 for (size_t j = 0; j < NL; ++j) {
222 U[i * N + j] = UL[i * NL + j];
223 VT[j * N + i] = VTL[j * NL + i];
224 }
225 }
226 for (size_t i = 0; i < NR; ++i) {
227 for (size_t j = 0; j < NR; ++j) {
228 U[(NL + i) * N + (NL + j)] = UR[i * NR + j];
229 VT[(NL + j) * N + (NL + i)] = VTR[j * NR + i];
230 }
231 }
232 }

238 template<typename T>
239 void bdsvd_dc(const T *d_in, const T *e_in, size_t n,
240 T *s, T *U, T *VT) {
241 if (n == 0) return;
242 if (n == 1) {
243 s[0] = std::abs(d_in[0]);
244 U[0] = T(1);
245 VT[0] = (d_in[0] >= T(0)) ? T(1) : T(-1);
246 return;
247 }
248 // Use QR-based bidiagonal SVD for n <= 256 (numerically stable).
249 // The D&C path is faster asymptotically but the merge step
250 // (solve_secular_equation_svd) is not yet numerically robust
251 // for ill-conditioned matrices.
252 if (n <= 256) {
253 bdsvd_qr(d_in, e_in, n, s, U, VT);
254 return;
255 }
256
257 size_t mid = n / 2;
258 std::vector<T> sL(mid), UL(mid * mid), VTL(mid * mid);
259 std::vector<T> sR(n - mid), UR((n - mid) * (n - mid)), VTR((n - mid) * (n - mid));
260
261 bdsvd_dc(d_in, e_in, mid, sL.data(), UL.data(), VTL.data());
262 bdsvd_dc(d_in + mid, e_in + mid, n - mid, sR.data(), UR.data(), VTR.data());
263
264 // Coupling vectors: last column of UL, first row of VTR
265 std::vector<T> z_coup(mid);
266 std::vector<T> w_coup(n - mid);
267 for (size_t i = 0; i < mid; ++i)
268 z_coup[i] = UL[i * mid + (mid - 1)];// last COLUMN of UL
269 for (size_t j = 0; j < n - mid; ++j)
270 w_coup[j] = VTR[j];// first ROW of VTR (VTR[0][j])
271
272 T rho = e_in[mid];
273
274 // Merge in block-diagonal order
275 merge_sorted_svd(sL.data(), sR.data(),
276 UL.data(), VTL.data(),
277 UR.data(), VTR.data(),
278 mid, n - mid, s, U, VT);
279
280 // Apply the divide-and-conquer SVD merge
281 // This solves the secular equation to find merged singular values
282 // and updates U and VT accordingly
283 apply_svd_merge(sL.data(), sR.data(),
284 z_coup.data(), w_coup.data(),
285 rho, mid, n - mid, U, VT, s);
286 }
