133 template<typename DType = DTypeDefault>
134 auto arange(DType start, DType stop, DType step = 1) {
135 NP_THROW_UNLESS(step != 0, "Step must not be zero.");
136
137 Size size = static_cast<Size>((stop - start) / step);
138 Shape shape{size};
139 NDArrayDynamic<DType> result{shape};
140 Size i{};
141 if (step > 0) {
142 for (DType value = start; value < stop; value += step) {
143 result.set(i++, value);
144 }
145 } else {
146 for (DType value = start; value > stop; value += step) {
147 result.set(i++, value);
148 }
149 }
150 return result;
151 }

198 template<typename DType = DTypeDefault>
199 auto linspace(DType start, DType stop, Size num = 50) {
200 NP_THROW_UNLESS(num > 0, "Number of samples must be non-negative.");
201
202 std::vector<DType> vector;
203 vector.resize(num);
204 const DType delta = (stop - start) / (static_cast<DType>(num) - 1);
205 if (delta == 0) {
206 NP_THROW_WITH_STACKTRACE(std::invalid_argument, "Invalid parameters, delta == 0");
207 }
208 size_t i = 0;
209 for (DType value = start; value <= stop; value += delta) {
210 vector[i++] = value;
211 }
212 return NDArrayDynamic<DType>{vector};
213 }

322 static constexpr Size kCudaRandMinSize = 500'000'000;
323
324 // Fast conversion of a 64-bit random value to double in [0,1).
325 // Uses the top 53 bits of the 64-bit value (the mantissa precision of double)
326 // and multiplies by 2^-53. This is significantly faster than
327 // std::uniform_real_distribution<double> which has branching and
328 // rejection sampling overhead.
329 //
330 // Result is in [0, 1) with uniform distribution across all 2^53 possible
331 // double values in that range (same as numpy's uniform distribution).
332 inline double fast_rand_double(std::uint64_t bits) {
333 return (bits >> 11) * 0x1.0p-53;
334 }
335
336 // Fast conversion of a 64-bit random value to float in [0,1).
337 // Uses the top 24 bits (the mantissa precision of float).
338 inline float fast_rand_float(std::uint64_t bits) {
339 return (bits >> 40) * 0x1.0p-24f;
340 }

353 template<typename DType = DTypeDefault>
354 auto rand(const Shape &shape, DType minValue = 0.0, DType maxValue = 1.0) {
355 auto size = shape.calcSizeByShape();
356 auto *data = new DType[size];
357
358#ifdef USE_CUDA
359 // Use GPU for very large arrays only (see kCudaRandMinSize)
360 if (size > kCudaRandMinSize) {
361 // Generate a random seed from the random device
362 unsigned long long seed = std::random_device{}();
363 internal::cuda::randUniform(data, size, minValue, maxValue, seed);
364 } else
365#endif
366 {
367 // Each thread uses its own thread_local generator to avoid:
368 // 1. Data races (std::mt19937_64 is not thread-safe)
369 // 2. Cache line contention from shared mutable state
370 // 3. Performance degradation at small array sizes
371#ifdef USE_OPENMP
372#pragma omp parallel default(none) shared(data, size, minValue, maxValue)
373#endif
374 {
375 auto &local_gen = getGenerator();
376 if constexpr (std::is_same_v<DType, double>) {
377 // Fast path for double: use inline bit conversion
378 // Avoids std::uniform_real_distribution overhead (branching, rejection sampling)
379 const double scale = (maxValue - minValue);
380 // index variable in OpenMP 'for' statement must have signed integral type
381#ifdef USE_OPENMP
382#pragma omp for
383#endif
384 for (std::int32_t offset = 0; offset < static_cast<std::int32_t>(size); ++offset) {
385 data[offset] = minValue + fast_rand_double(local_gen()) * scale;
386 }
387 } else if constexpr (std::is_same_v<DType, float>) {
388 // Fast path for float: use inline bit conversion
389 const float scale = static_cast<float>(maxValue - minValue);
390#ifdef USE_OPENMP
391#pragma omp for
392#endif
393 for (std::int32_t offset = 0; offset < static_cast<std::int32_t>(size); ++offset) {
394 data[offset] = static_cast<float>(minValue) + fast_rand_float(local_gen()) * scale;
395 }
396 } else {
397 // Generic path for other types (e.g., int)
398 std::uniform_real_distribution<DType> distribution(minValue, maxValue);
399#ifdef USE_OPENMP
400#pragma omp for
401#endif
402 for (std::int32_t offset = 0; offset < static_cast<std::int32_t>(size); ++offset) {
403 data[offset] = distribution(local_gen);
404 }
405 }
406 }
407 }
408
409 return NDArrayDynamic<DType>{data, shape};
410 }

429 template<typename DType, Size SizeT, Size... Sizes>
430 struct rand_helper {
431 static Size constexpr m_size = (SizeT * ... * Sizes);
432
433 rand_helper() {
434 auto &local_gen = getGenerator();
435 std::vector<DType> vector;
436 vector.resize(m_size);
437 if constexpr (std::is_same_v<DType, double>) {
438 std::generate(vector.begin(), vector.end(), [&] { return fast_rand_double(local_gen()); });
439 } else if constexpr (std::is_same_v<DType, float>) {
440 std::generate(vector.begin(), vector.end(), [&] { return fast_rand_float(local_gen()); });
441 } else {
442 std::uniform_real_distribution<DType> distribution;
443 std::generate(vector.begin(), vector.end(), [&] { return distribution(local_gen); });
444 }
445 const Shape shape{SizeT, Sizes...};
446 m_array = NDArrayStatic<DType, m_size>(vector, shape);
447 }
448
449 explicit operator NDArrayStatic<DType, m_size>() {
450 return m_array;
451 }
452
453 NDArrayStatic<DType, m_size> m_array;
454 };
