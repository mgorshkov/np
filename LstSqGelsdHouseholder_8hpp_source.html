48 template<typename T>
67 static inline T householder_generate(T *x, size_t n, T *alpha_out = nullptr) {
68 if (n == 0) return T(0);
69 if (n == 1) {
70 T alpha = x[0];
71 if (alpha_out) *alpha_out = alpha;
72 x[0] = T(1);
73 return T(0);
74 }
75
76 T x0 = x[0];
77 T sigma = T(0);
78 for (size_t i = 1; i < n; ++i)
79 sigma += x[i] * x[i];
80
81 // alpha = -sign(x0) * sqrt(x0^2 + sigma) (LAPACK's beta)
82 // LAPACK DLARFG: beta = -sign(x0_in) * sqrt(x0_in^2 + sigma)
83 // If x0 >= 0: alpha = -sqrt(...) (negative)
84 // If x0 < 0: alpha = +sqrt(...) (positive)
85 T alpha = std::sqrt(x0 * x0 + sigma);
86 if (x0 >= T(0)) alpha = -alpha;
87 if (alpha_out) *alpha_out = alpha;
88
89 // beta = alpha - x0 (LAPACK's beta - x0_in)
90 T beta = alpha - x0;
91
92 // Check for zero norm (x is already on the axis)
93 if (beta == T(0) && sigma == T(0)) {
94 x[0] = T(1);
95 return T(0);
96 }
97
98 // Store reflector: v[0] = 1 (implicit)
99 // v[1..n-1] = x[1..n-1] / (x0 - alpha) = -x[1..n-1] / beta
100 // This matches LAPACK DLARFG: x[i] = x[i] / (x0_in - beta_lapack)
101 T inv_scale = T(1) / (x0 - alpha);// = -1/beta
102 x[0] = T(1);
103 for (size_t i = 1; i < n; ++i)
104 x[i] *= inv_scale;
105
106 return beta / alpha;
107 }

118 template<typename T>
119 static inline void householder_apply_left(T tau, const T *v,
120 T *C, size_t m, size_t n, size_t ldc) {
121 if (tau == T(0) || m == 0 || n == 0) return;
122
123 // w[j] = sum_{i=0..m-1} v[i] * C[i, j]
124 for (size_t j = 0; j < n; ++j) {
125 T w = T(0);
126 for (size_t i = 0; i < m; ++i)
127 w += v[i] * C[i * ldc + j];
128 w *= tau;
129 for (size_t i = 0; i < m; ++i)
130 C[i * ldc + j] -= w * v[i];
131 }
132 }

139 template<typename T>
140 static inline void householder_apply_right(T tau, const T *v,
141 T *C, size_t m, size_t n, size_t ldc) {
142 if (tau == T(0) || m == 0 || n == 0) return;
143
144 // w[i] = sum_{j=0..n-1} C[i, j] * v[j]
145 for (size_t i = 0; i < m; ++i) {
146 T w = T(0);
147 for (size_t j = 0; j < n; ++j)
148 w += C[i * ldc + j] * v[j];
149 w *= tau;
150 for (size_t j = 0; j < n; ++j)
151 C[i * ldc + j] -= w * v[j];
152 }
153 }

160 template<typename T>
161 static inline void gemm_nn_add(T *C, size_t ldc,
162 const T *A, size_t lda,
163 const T *B, size_t ldb,
164 size_t m, size_t n, size_t k) {
165 constexpr size_t MC = 64;
166 constexpr size_t NC = 256;
167 constexpr size_t KC = 256;
168
169 for (size_t mc = 0; mc < m; mc += MC) {
170 size_t mr = std::min(MC, m - mc);
171 for (size_t nc = 0; nc < n; nc += NC) {
172 size_t nr = std::min(NC, n - nc);
173 for (size_t kc = 0; kc < k; kc += KC) {
174 size_t kr = std::min(KC, k - kc);
175 for (size_t i = 0; i < mr; ++i) {
176 size_t row = mc + i;
177 for (size_t j = 0; j < nr; ++j) {
178 T sum = T(0);
179 for (size_t p = 0; p < kr; ++p)
180 sum += A[row * lda + (kc + p)] * B[(kc + p) * ldb + (nc + j)];
181 C[row * ldc + (nc + j)] += sum;
182 }
183 }
184 }

368 template<typename T>
369 static inline void larft(const T *Y, size_t m, size_t NB,
370 const T *tau, size_t ldy,
371 T *T_, size_t ldT) {
372 if (NB == 0) return;
373
374 // T[0,0] = tau[0]
375 T_[0] = tau[0];
376
377 for (size_t j = 1; j < NB; ++j) {
378 // Step 1: w[i] = Y[:,i]^T * Y[:,j] for i = 0..j-1
379 // Y is column-major: Y[k + i*ldy] is element (k,i)
380 // Store w_i temporarily in T[j, i] (we'll overwrite later)
381 for (size_t i = 0; i < j; ++i) {
382 T w_i = T(0);
383 for (size_t k = 0; k < m; ++k)
384 w_i += Y[k + i * ldy] * Y[k + j * ldy];
385 T_[i + j * ldT] = w_i;
386 }
387
388 // Step 2: w = T[0:j,0:j] * w (matrix-vector multiply, DTRMV equivalent)
389 // T is upper-triangular column-major: T[i,k] = T_[i + k*ldT]
390 // T[i,k] = 0 for k < i (upper triangular).
391 // w_new[i] = sum_{k=i..j-1} T[i,k] * w[k]
392 for (size_t i = 0; i < j; ++i) {
393 T sum = T(0);
394 for (size_t k = i; k < j; ++k)
395 sum += T_[i + k * ldT] * T_[k + j * ldT];
396 T_[i + j * ldT] = sum;
397 }
398
399 // Step 3: T[0:j, j] = -tau[j] * x
400 T tau_j = tau[j];
401 for (size_t i = 0; i < j; ++i)
402 T_[i + j * ldT] *= -tau_j;
403
404 // Step 4: T[j, j] = tau[j]
405 T_[j + j * ldT] = tau_j;
406 }
407 }

409 // ---------------------------------------------------------------
410 // LAPACK-style blocked application (DLARFB equivalent)
411 // ---------------------------------------------------------------
412 //
413 // LAPACK's DORMBR/DORMQR uses DGEMM for the Y^T*C and Y*W products,
414 // and DTRMM for the T*W product. We implement the same approach
415 // with explicit cache tiling in the GEMM operations.
416 //
417 // The key difference from the naive triple-loop implementation:
418 // - Cache-friendly tiling in the m and n dimensions
419 // - Better memory access patterns (contiguous inner loops)
420 // - Reduced TLB misses for large matrices
421
433 template<typename T>
434 static inline void larfb_left(const T *Y, size_t m, size_t NB,
435 const T *T_, size_t ldT,
436 T *C, size_t n, size_t ldc,
437 size_t ldy = 0,
438 T *W_buf = nullptr) {
439 if (ldy == 0) ldy = m;
440 if (NB == 0 || n == 0) return;
441
442 // LAPACK approach:
443 // W = Y^T * C (NB x n) — GEMM (Y^T is NB x m, C is m x n)
444 // W = T * W (NB x n) — TRMM (T is NB x NB upper-tri)
445 // C = C - Y * W (m x n) — GEMM (Y is m x NB, W is NB x n)
446
447 // Use pre-allocated buffer if provided, otherwise allocate locally
448 std::vector<T> W_local;
449 T *W = W_buf;
450 if (W == nullptr) {
451 W_local.resize(NB * n, T(0));
452 W = W_local.data();
453 } else {
454 // Zero the work buffer
455 std::memset(W, 0, NB * n * sizeof(T));
456 }
457
458 // Step 1: W = Y^T * C (NB x n)
459 // Tile over m (the reduction dimension) for cache efficiency.
460 {
461 constexpr size_t KC = 256;
462 for (size_t kc = 0; kc < m; kc += KC) {
463 size_t kr = std::min(KC, m - kc);
464 for (size_t i = 0; i < NB; ++i) {
465 for (size_t j = 0; j < n; ++j) {
466 T sum = T(0);
467 for (size_t k = 0; k < kr; ++k)
468 sum += Y[(kc + k) + i * ldy] * C[(kc + k) * ldc + j];
469 W[i * n + j] += sum;
470 }
471 }
472 }
473 }
474
475 // Step 2: W = T * W (NB x n), T is NB x NB upper-triangular column-major
476 for (size_t j = 0; j < n; ++j) {
477 for (size_t i = 0; i < NB; ++i) {
478 T sum = T(0);
479 for (size_t k = i; k < NB; ++k)
480 sum += T_[i + k * ldT] * W[k * n + j];
481 W[i * n + j] = sum;
482 }
483 }
484
485 // Step 3: C = C - Y * W (m x n)
486 // Tile over m and n.
487 {
488 constexpr size_t MC = 64;
489 constexpr size_t NC = 256;
490 for (size_t mc = 0; mc < m; mc += MC) {
491 size_t mr = std::min(MC, m - mc);
492 for (size_t nc = 0; nc < n; nc += NC) {
493 size_t nr = std::min(NC, n - nc);
494 for (size_t i = 0; i < mr; ++i) {
495 size_t row = mc + i;
496 for (size_t j = 0; j < nr; ++j) {
497 T sum = T(0);
498 for (size_t p = 0; p < NB; ++p)
499 sum += Y[row + p * ldy] * W[p * n + (nc + j)];
500 C[row * ldc + (nc + j)] -= sum;
501 }
502 }
503 }
504 }
505 }

519 template<typename T>
520 static inline void larfb_right(const T *Y, size_t n, size_t NB,
521 const T *T_, size_t ldT,
522 T *C, size_t m, size_t ldc,
523 size_t ldy = 0,
524 T *W_buf = nullptr) {
525 if (ldy == 0) ldy = n;
526 if (NB == 0 || m == 0) return;
527
528 // LAPACK approach:
529 // W = C * Y (m x NB) — GEMM (C is m x n, Y is n x NB)
530 // W = W * T (m x NB) — TRMM (T is NB x NB upper-tri)
531 // C = C - W * Y^T (m x n) — GEMM (W is m x NB, Y^T is NB x n)
532
533 // Use pre-allocated buffer if provided, otherwise allocate locally
534 std::vector<T> W_local;
535 T *W = W_buf;
536 if (W == nullptr) {
537 W_local.resize(m * NB, T(0));
538 W = W_local.data();
539 } else {
540 std::memset(W, 0, m * NB * sizeof(T));
541 }
542
543 // Step 1: W = C * Y (m x NB)
544 // Tile over n (the reduction dimension).
545 {
546 constexpr size_t KC = 256;
547 for (size_t kc = 0; kc < n; kc += KC) {
548 size_t kr = std::min(KC, n - kc);
549 for (size_t p = 0; p < m; ++p) {
550 for (size_t i = 0; i < NB; ++i) {
551 T sum = T(0);
552 for (size_t j = 0; j < kr; ++j)
553 sum += C[p * ldc + (kc + j)] * Y[(kc + j) + i * ldy];
554 W[p * NB + i] += sum;
555 }
556 }
557 }
558 }
559
560 // Step 2: W = W * T (m x NB), T is NB x NB upper-triangular column-major
561 for (size_t p = 0; p < m; ++p) {
562 for (size_t i = NB; i > 0;) {
563 --i;
564 T sum = T(0);
565 for (size_t k = 0; k <= i; ++k)
566 sum += W[p * NB + k] * T_[k + i * ldT];
567 W[p * NB + i] = sum;
568 }
569 }
570
571 // Step 3: C = C - W * Y^T (m x n)
572 // Tile over m and n.
573 {
574 constexpr size_t MC = 64;
575 constexpr size_t NC = 256;
576 for (size_t mc = 0; mc < m; mc += MC) {
577 size_t mr = std::min(MC, m - mc);
578 for (size_t nc = 0; nc < n; nc += NC) {
579 size_t nr = std::min(NC, n - nc);
580 for (size_t i = 0; i < mr; ++i) {
581 size_t row = mc + i;
582 for (size_t j = 0; j < nr; ++j) {
583 T sum = T(0);
584 for (size_t p = 0; p < NB; ++p)
585 sum += W[row * NB + p] * Y[(nc + j) + p * ldy];
586 C[row * ldc + (nc + j)] -= sum;
587 }
588 }
589 }
590 }
