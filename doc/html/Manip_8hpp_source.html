249 template<typename DType, typename Derived, typename Storage, typename... Args>
250 auto column_stack(const ndarray::internal::NDArrayBase<DType, Derived, Storage> &arg1, Args &&...args) {
251 if (allEmpty(arg1, std::forward<Args>(args)...)) {
252 return ndarray::array_dynamic::NDArrayDynamic<DType>{};
253 }
254
255 auto maxDims = getMaxDims(arg1, std::forward<Args>(args)...);
256 if (maxDims == 1) {
257 maxDims = 2;
258 }
259 Shape shape{};
260 for (Size dim = 0; dim < maxDims; ++dim) {
261 Size size{};
262 if (dim == 1) {
263 size = getSumDim(dim, arg1, std::forward<Args>(args)...);
264 } else {
265 size = checkDim(dim, arg1, std::forward<Args>(args)...);
266 }
267 shape.addDim(size);
268 }
269
270 ndarray::array_dynamic::NDArrayDynamic<DType> result{shape};
271 Size indexTarget = 0;
272 for (Size rowSource = 0; rowSource < shape[0]; ++rowSource) {
273 fill(result, indexTarget, rowSource, arg1, std::forward<Args>(args)...);
274 }
275 return result;
276 }

317 auto stack(const ndarray::internal::NDArrayBase<DType, Derived, Storage> &arg1, Args &&...args) {
318 if (allEmpty(arg1, std::forward<Args>(args)...)) {
319 return ndarray::array_dynamic::NDArrayDynamic<DType>{};

324 Size indexTarget = 0;

400 //
410 // Template callable overload (avoids std::function virtual dispatch overhead).
411 // Enabled only when Cond/Pos/Neg are not std::function to avoid ambiguity.
412 template<typename DType, typename Derived, typename Storage,
413 typename Cond, typename Pos, typename Neg,
414 typename = std::enable_if_t<
415 !std::is_same_v<std::decay_t<Cond>, std::function<bool(const DType &)>> ||
416 !std::is_same_v<std::decay_t<Pos>, std::function<DType(const DType &)>> ||
417 !std::is_same_v<std::decay_t<Neg>, std::function<DType(const DType &)>>>>
418 auto where(const ndarray::internal::NDArrayBase<DType, Derived, Storage> &x,
419 Cond &&condition,
420 Pos &&positive,
421 Neg &&negative) {
422 NDArrayDynamic<DType> result{x.shape()};
423 auto *result_data = result.data();
424 auto n = x.size();
425 if constexpr (Storage::is_contiguous) {
426 // Fast path: direct pointer access avoids virtual get()/set() dispatch
427 const auto *src = x.data();
428 // Use SIMD-accelerated path for double/float contiguous arrays
429 if constexpr (std::is_same_v<DType, double>) {
430 if (np::internal::simd_at_least(np::internal::SimdLevel::AVX2)) {
431 for (Size i = 0; i < n; ++i) {
432 result_data[i] = condition(src[i]) ? positive(src[i]) : negative(src[i]);
433 }
434 } else {
435 for (Size i = 0; i < n; ++i) {
436 result_data[i] = condition(src[i]) ? positive(src[i]) : negative(src[i]);
437 }
438 }
439 } else {
440 for (Size i = 0; i < n; ++i) {
441 result_data[i] = condition(src[i]) ? positive(src[i]) : negative(src[i]);
442 }
443 }
444 } else {
445 // Slow path: element-by-element via virtual dispatch
446 for (Size i = 0; i < n; ++i) {
447 const auto &element = x.get(i);
448 result_data[i] = condition(element) ? positive(element) : negative(element);
449 }
450 }
451 return result;
452 }

454 // In-place where() variant: writes directly into an existing output array.
455 // Avoids allocation overhead for the IRLS Tukey bisquare weight update pattern.
456 // The output array must have the same size as the input.
457 template<typename DType, typename Derived, typename Storage,
458 typename DerivedOut, typename StorageOut,
459 typename Cond, typename Pos, typename Neg,
460 typename = std::enable_if_t<
461 !std::is_same_v<std::decay_t<Cond>, std::function<bool(const DType &)>> ||
462 !std::is_same_v<std::decay_t<Pos>, std::function<DType(const DType &)>> ||
463 !std::is_same_v<std::decay_t<Neg>, std::function<DType(const DType &)>>>>
464 void where_inplace(const ndarray::internal::NDArrayBase<DType, Derived, Storage> &x,
465 ndarray::internal::NDArrayBase<DType, DerivedOut, StorageOut> &out,
466 Cond &&condition,
467 Pos &&positive,
468 Neg &&negative) {
469 auto n = x.size();
470 if constexpr (Storage::is_contiguous && StorageOut::is_contiguous) {
471 // Fast path: direct pointer access for both input and output
472 const auto *src = x.data();
473 auto *dst = out.data();
474 for (Size i = 0; i < n; ++i) {
475 dst[i] = condition(src[i]) ? positive(src[i]) : negative(src[i]);
476 }
477 } else {
478 // Fallback: element-by-element via virtual dispatch
479 for (Size i = 0; i < n; ++i) {
480 const auto &element = x.get(i);
481 out.set(i, condition(element) ? positive(element) : negative(element));
482 }
483 }
484 }

486 // Specialized Tukey bisquare weight update with SIMD acceleration.
487 // Computes: result[i] = (a[i] <= k) ? 1.0 : (2*k/a[i] - k*k/(a[i]*a[i]))
488 // This is the hot path in IRLS robust regression loops.
489 template<typename DType, typename Derived, typename Storage>
490 auto where_tukey(const ndarray::internal::NDArrayBase<DType, Derived, Storage> &x, DType k) {
491 NDArrayDynamic<DType> result{x.shape()};
492 auto *result_data = result.data();
493 auto n = x.size();
494 if constexpr (Storage::is_contiguous) {
495 const auto *src = x.data();
496 if constexpr (std::is_same_v<DType, double>) {
497 np::internal::where_tukey_pd(src, static_cast<double>(k), result_data, n);
498 } else if constexpr (std::is_same_v<DType, float>) {
499 np::internal::where_tukey_ps(src, static_cast<float>(k), result_data, n);
500 } else {
501 for (Size i = 0; i < n; ++i) {
502 result_data[i] = (src[i] <= k) ? DType(1) : (DType(2) * k / src[i] - k * k / (src[i] * src[i]));
503 }
504 }
505 } else {
506 for (Size i = 0; i < n; ++i) {
507 const auto &element = x.get(i);
508 result.set(i, (element <= k) ? DType(1) : (DType(2) * k / element - k * k / (element * element)));
509 }
510 }
511 return result;
512 }

515 template<typename DType, typename Derived, typename Storage,
516 typename DerivedOut, typename StorageOut>
517 void where_tukey_inplace(const ndarray::internal::NDArrayBase<DType, Derived, Storage> &x,
518 ndarray::internal::NDArrayBase<DType, DerivedOut, StorageOut> &out,

541 // Fused abs(a - b): computes abs(a[i] - b[i]) in a single SIMD pass.
542 // Avoids intermediate array allocation that would occur with abs(a - b).
543 template<typename DType, typename Derived1, typename Storage1, typename Derived2, typename Storage2>
544 auto abs_sub(const ndarray::internal::NDArrayBase<DType, Derived1, Storage1> &a,
545 const ndarray::internal::NDArrayBase<DType, Derived2, Storage2> &b) {
546 NDArrayDynamic<DType> result{a.shape()};
547 auto *result_data = result.data();
548 auto n = a.size();
549 if constexpr (Storage1::is_contiguous && Storage2::is_contiguous) {
550 const auto *a_data = a.data();
551 const auto *b_data = b.data();
552 if constexpr (std::is_same_v<DType, double>) {
553 np::internal::abs_sub_pd(a_data, b_data, result_data, n);
554 } else if constexpr (std::is_same_v<DType, float>) {
555 np::internal::abs_sub_ps(a_data, b_data, result_data, n);
556 } else {
557 for (Size i = 0; i < n; ++i) {
558 result_data[i] = std::abs(a_data[i] - b_data[i]);
559 }
560 }
561 } else {
562 for (Size i = 0; i < n; ++i) {
563 result.set(i, std::abs(a.get(i) - b.get(i)));
564 }
565 }
566 return result;
567 }

569 // Fused sum(a * a * w): computes sum(a[i]*a[i]*w[i]) in a single SIMD pass.
570 // Avoids intermediate array allocation that would occur with sum((a * a * w)).
571 template<typename DType, typename Derived1, typename Storage1, typename Derived2, typename Storage2>
572 auto sum_sq_weighted(const ndarray::internal::NDArrayBase<DType, Derived1, Storage1> &a,
573 const ndarray::internal::NDArrayBase<DType, Derived2, Storage2> &w) {
574 auto n = a.size();
575 if constexpr (Storage1::is_contiguous && Storage2::is_contiguous) {
576 const auto *a_data = a.data();
577 const auto *w_data = w.data();
578 if constexpr (std::is_same_v<DType, double>) {
579 return np::internal::sum_sq_weighted_pd(a_data, w_data, n);
580 } else if constexpr (std::is_same_v<DType, float>) {
581 return static_cast<double>(np::internal::sum_sq_weighted_ps(a_data, w_data, n));
582 } else {
583 DType result{};
584 for (Size i = 0; i < n; ++i) {
585 result += a_data[i] * a_data[i] * w_data[i];
586 }
587 return static_cast<double>(result);
588 }
589 } else {
590 double result{};
591 for (Size i = 0; i < n; ++i) {
592 auto av = a.get(i);
593 result += av * av * w.get(i);
594 }
595 return result;
596 }
