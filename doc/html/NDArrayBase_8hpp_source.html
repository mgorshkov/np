218 template<typename DType2, typename Derived2, typename Storage2>
219 bool array_equal(const NDArrayBase<DType2, Derived2, Storage2> &array) const {
220 auto shape1 = shape();
221 auto shape2 = array.shape();
222 if (shape1 != shape2)
223 return false;
224 for (Size index = 0; index < size(); ++index) {
225 if (!np::internal::element_equal(get(index), array.get(index)))
226 return false;
227 }
228 return true;
229 }

382 // For 2D arrays
383 const DType &at(Size row, Size column) const {
384 if (shape().size() != 2) {
385 NP_THROW_WITH_STACKTRACE(std::invalid_argument, "At method is supported for 2D arrays only");
386 }
387 if (row >= shape()[0]) {
388 NP_THROW_WITH_STACKTRACE(std::invalid_argument, "Invalid row");
389 }
390 if (column >= shape()[1]) {
391 NP_THROW_WITH_STACKTRACE(std::invalid_argument, "Invalid column");
392 }
393 return m_storage.get(row * shape()[1] + column);
394 }

502 struct NDArrayBaseHasher {
503 template<typename DType, typename Derived, typename Storage>
504 auto operator()(const NDArrayBase<DType, Derived, Storage> &array) const -> size_t {
505 size_t h{0};
506 for (Size i = 0; i < array.size(); ++i) {
507 h ^= std::hash<DType>{}(array.get(i));
508 }
509 return h;
